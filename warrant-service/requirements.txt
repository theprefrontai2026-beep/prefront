# Prefront/Warrant PDS service: the enforcement plane over HTTP.
#
# The ENGINE (`warrant/`) deliberately depends on cryptography alone, because a
# resource server verifying a Mission offline must not have to install a web
# framework. This package is where wire concerns live, so it is free to take
# the repo's usual FastAPI stack — and keeping the split means the engine stays
# embeddable in a process that has no interest in HTTP at all.
fastapi>=0.110
uvicorn[standard]>=0.29
pydantic>=2.6
pyyaml>=6.0
cryptography>=42.0
# JWT verification for subject tokens (oidc.py). Deliberately a library rather
# than hand-rolled: algorithm confusion, unverified `alg`, `none` and key-type
# substitution are all live foot-guns, and this is the canonical place not to
# be clever.
pyjwt[crypto]>=2.8
